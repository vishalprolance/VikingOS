22/// Signature for the entrypoint on processor hand-over from the bootloader.
23pub type MpGotoFunction = unsafe extern "C" fn(&MpInfo) -> !;
24
25impl MpInfo {
26    /// Start this processor.
27    /// Writes the extra argument to the struct then writes the goto pointer, triggering the CPU to start running the function provided.
28    pub fn bootstrap(&self, address: MpGotoFunction, extra_arg: u64) {
29        use core::sync::atomic::Ordering;
30
31        self.extra_argument.store(extra_arg, Ordering::Relaxed);
32        self.goto_addr.store(address as *mut (), Ordering::Release);
33    }
34
35    /// Retrieve the extra argument passed to this processor by [`Self::bootstrap`].
36    pub fn extra_argument(&self) -> u64 {
37        use core::sync::atomic::Ordering;
38
39        self.extra_argument.load(Ordering::Acquire)
40    }
41}
42
43impl MpRespData {
44    pub const fn cpus(&self) -> &[&MpInfo] {
45        unsafe { &*core::ptr::slice_from_raw_parts(self.cpus as _, self.cpu_count as usize) }
46    }
47}
