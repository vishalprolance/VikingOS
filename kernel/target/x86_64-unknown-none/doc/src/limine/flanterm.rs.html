9#[repr(C)]
10pub struct FlantermParams {
11    canvas: *mut u8,
12    canvas_size: u64,
13    pub ansi_colors: [u32; 8],
14    pub ansi_bright_colors: [u32; 8],
15    pub default_bg: u32,
16    pub default_fg: u32,
17    pub default_bg_bright: u32,
18    pub default_fg_bright: u32,
19    font: *mut (),
20    font_width: u64,
21    font_height: u64,
22    font_spacing: u64,
23    font_scale_x: u64,
24    font_scale_y: u64,
25    pub margin: u64,
26    pub rotation: u64,
27}
28
29#[repr(C)]
30pub struct Font<'a> {
31    pub font: &'a [u8],
32    pub width: u64,
33    pub height: u64,
34    pub spacing: u64,
35    pub scale_x: u64,
36    pub scale_y: u64,
37}
38
39impl FlantermParams {
40    pub const fn canvas(&self) -> Option<&[u8]> {
41        if self.canvas.is_null() {
42            return None;
43        }
44        Some(unsafe { &*core::ptr::slice_from_raw_parts(self.canvas, self.canvas_size as usize) })
45    }
46
47    pub const fn canvas_mut(&mut self) -> &mut [u8] {
48        unsafe { &mut *core::ptr::slice_from_raw_parts_mut(self.canvas, self.canvas_size as usize) }
49    }
50
51    const fn font_size(&self) -> usize {
52        self.font_width as usize * self.font_height as usize * 256 / 8
53    }
54
55    pub const fn font<'a>(&'a self) -> Option<Font<'a>> {
56        if self.font.is_null() {
57            return None;
58        }
59        Some(unsafe {
60            Font {
61                font: &*core::ptr::slice_from_raw_parts(self.font as *const u8, self.font_size()),
62                width: self.font_width,
63                height: self.font_height,
64                spacing: self.font_spacing,
65                scale_x: self.font_scale_x,
66                scale_y: self.font_scale_y,
67            }
68        })
69    }
70}
