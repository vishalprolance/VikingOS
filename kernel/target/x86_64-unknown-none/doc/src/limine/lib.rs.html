27/// Request that Limine uses a certain base revision.
28/// If omitted, 0 is used.
29/// A conformant kernel should test that either [`BaseRevision::is_supported`] is true,
30/// or [`BaseRevision::actual_revision()`] specifies a revision supported by the kernel.
31#[repr(C)]
32pub struct BaseRevision {
33    magic: UnsafeCell<[u64; 3]>,
34}
35unsafe impl Send for BaseRevision {}
36unsafe impl Sync for BaseRevision {}
37
38impl BaseRevision {
39    /// Currently highest supported base revision.
40    pub const MAX_SUPPORTED: u64 = 6;
41
42    /// Use the default base revision.
43    /// Identical to `BaseRevision::with_revision(BaseRevision::MAX_SUPPORTED)`.
44    pub const fn new() -> Self {
45        Self::with_revision(Self::MAX_SUPPORTED)
46    }
47
48    /// Use a specific base revision.
49    pub const fn with_revision(revision: u64) -> Self {
50        Self {
51            magic: UnsafeCell::new([0xf9562b2d5c95a6c8, 0x6a7b384944536bdc, revision]),
52        }
53    }
54
55    /// Whether the requested revision is supported.
56    pub fn is_supported(&self) -> bool {
57        unsafe { (self.magic.get() as *const u64).add(2).read_volatile() == 0 }
58    }
59
60    /// What revision is actually in use right now, regardless of whether it is the requested one.
61    pub fn actual_revision(&self) -> Option<u64> {
62        let actual = unsafe { (self.magic.get() as *const u64).add(1).read_volatile() };
63        if actual == 0x6a7b384944536bdc {
64            None
65        } else {
66            Some(actual)
67        }
68    }
69}
