1// VikingOS Kernel — Virtual File System (VFS)
2//
3// The VFS layer provides a unified interface for accessing different
4// types of filesystems, abstracting away the underlying storage.
5
6use alloc::vec::Vec;
7use crate::storage::StorageError;
8
9/// Error types for VFS operations.
10#[derive(Debug)]
11pub enum VfsError {
12    NotFound,
13    PermissionDenied,
14    StorageError(StorageError),
15    InvalidPath,
16    NotADirectory,
17}
18
19/// A unique identifier for a file or directory within a filesystem.
20#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
21pub struct Inode(pub u64);
22
23/// Metadata for a file or directory.
24pub struct InodeMetadata {
25    pub size: u64,
26    pub is_directory: bool,
27    pub permissions: u32,
28}
29
30/// A directory entry mapping a name to an inode.
31pub struct DirEntry {
32    pub name: alloc::string::String,
33    pub inode: Inode,
34}
35
36/// The core trait that any filesystem implementation must provide.
37pub trait FileSystem: Send + Sync {
38    /// Read data from an inode at a specific offset.
39    fn read(&self, inode: Inode, offset: u64, buf: &mut [u8]) -> Result<usize, VfsError>;
40
41    /// Write data to an inode at a specific offset.
42    fn write(&self, inode: Inode, offset: u64, buf: &[u8]) -> Result<usize, VfsError>;
43
44    /// Look up an inode by name within a parent directory.
45    fn lookup(&self, parent: Inode, name: &str) -> Result<Inode, VfsError>;
46
47    /// Read the contents of a directory.
48    fn readdir(&self, inode: Inode) -> Result<Vec<DirEntry>, VfsError>;
49
50    /// Get metadata for a specific inode.
51    fn get_metadata(&self, inode: Inode) -> Result<InodeMetadata, VfsError>;
52}
53
54/// A handle to an open file, tracking the current offset.
55pub struct FileHandle {
56    pub fs: &'static dyn FileSystem,
57    pub inode: Inode,
58    pub offset: u64,
59}
60
61impl FileHandle {
62    pub fn read(&mut self, buf: &mut [u8]) -> Result<usize, VfsError> {
63        let bytes_read = self.fs.read(self.inode, self.offset, buf)?;
64        self.offset += bytes_read as u64;
65        Ok(bytes_read)
66    }
67
68    pub fn write(&mut self, buf: &[u8]) -> Result<usize, VfsError> {
69        let bytes_written = self.fs.write(self.inode, self.offset, buf)?;
70        self.offset += bytes_written as u64;
71        Ok(bytes_written)
72    }
73}
