1// VikingOS Kernel — Local APIC (LAPIC) Driver
2//
3// The Local APIC manages interrupts for the local CPU core.
4// This driver handles initialization, EOI signaling, and timer configuration.
5
6#![allow(dead_code)]
7
8use crate::arch::io::{outl, inl};
9use crate::arch::interrupts;
10
11/// LAPIC base address (standard for x86-64).
12/// In a real system, this should be verified via ACPI MADT, but 0xFEE00000 is the standard.
13pub const LAPIC_BASE: usize = 0xFEE00000;
14
15// --- LAPIC Register Offsets ---
16const REG_ID: u16 = 0x00;
17const REG_SVR: u16 = 0x0F0;
18const REG_ICR_LOW: u16 = 0x300;
19const REG_ICR_HIGH: u16 = 0x310;
20const REG_LVT_TIMER: u16 = 0x320;
21const REG_TIMER_INIT_COUNT: u16 = 0x380;
22const REG_TIMER_CUR_COUNT: u16 = 0x390;
23
24/// Spurious Interrupt Vector.
25/// Must be set to a vector that is not used by other hardware interrupts.
26const SPURIOUS_VECTOR: u8 = 0xFF;
27
28pub struct LocalApic {
29    base: usize,
30}
31
32impl LocalApic {
33    /// Create a new LAPIC handle.
34    pub const fn new() -> Self {
35        Self {
36            base: LAPIC_BASE,
37        }
38    }
39
40    /// Read a 32-bit value from a LAPIC register.
41    unsafe fn read_reg(&self, offset: u16) -> u32 {
42        // LAPIC registers are memory-mapped, not I/O mapped.
43        // We use volatile reads to ensure the compiler doesn't optimize them away.
44        core::ptr::read_volatile((self.base + offset as usize) as *const u32)
45    }
46
47    /// Write a 32-bit value to a LAPIC register.
48    unsafe fn write_reg(&self, offset: u16, value: u32) {
49        core::ptr::write_volatile((self.base + offset as usize) as *mut u32, value);
50    }
51
52    /// Initialize the LAPIC.
53    pub fn init(&self) {
54        unsafe {
55            // Set the Spurious Interrupt Vector Register (SVR).
56            // Bit 8 must be set to enable the LAPIC.
57            let svr_val = (SPURIOUS_VECTOR as u32) | (1 << 8);
58            self.write_reg(REG_SVR, svr_val);
59        }
60    }
61
62    /// Send an End of Interrupt (EOI) signal to the LAPIC.
63    /// This tells the LAPIC that the current interrupt has been handled.
64    pub fn send_eoi(&self) {
65        unsafe {
66            self.write_reg(0, 0); // EOI register is at offset 0
67        }
68    }
69
70    /// Configure the LAPIC Timer.
71    ///
72    /// # Arguments
73    /// * `vector` - The IDT vector to trigger on timer expiration.
74    /// * `frequency_hz` - The desired tick rate (e.g., 100 for 100Hz).
75    pub fn setup_timer(&self, vector: u8, frequency_hz: u32) {
76        unsafe {
77            // 1. Set the LVT Timer register:
78            // - Periodic mode (bit 17 = 1)
79            // - Vector in bits 0-7
80            let lvt_val = (vector as u32) | (1 << 17);
81            self.write_reg(REG_LVT_TIMER, lvt_val);
82
83            // 2. Calculate divisor.
84            // APIC timer frequency is usually the bus clock / 2 or 3.
85            // For this implementation, we'll assume a standard 10MHz clock for simplicity,
86            // and we'll set the initial count.
87            // Initial Count = Clock / frequency
88            let count = 10_000_000 / frequency_hz;
89            self.write_reg(REG_TIMER_INIT_COUNT, count);
90        }
91    }
92}
93
94/// Global LAPIC instance.
95pub static LAPIC: LocalApic = LocalApic::new();
