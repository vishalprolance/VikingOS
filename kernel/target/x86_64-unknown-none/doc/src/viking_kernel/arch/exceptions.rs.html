1// VikingOS Kernel — x86-64 CPU Exception Handlers & Crash Dump Architecture
2//
3// Intercepts all 32 standard x86-64 CPU exceptions. Each exception has an
4// assembly trampoline stub that pushes an aligned, uniform stack frame
5// before jumping into the Rust exception dispatcher.
6//
7// For non-fatal traps (e.g., Breakpoint #BP), the handler logs the event and
8// returns to let execution proceed. For fatal faults (e.g., Page Fault,
9// Double Fault, General Protection Fault), a formatted crash dump with all
10// register states is printed to the serial console before halting the CPU.
11

144    // Pass pointer to ExceptionContext (which starts at current RSP)

166    pop r13

168    pop r15

171    add rsp, 16
172

174    iretq

228/// Register all 32 standard CPU exception handlers in the IDT.
229pub fn register_all(idt: &mut InterruptDescriptorTable) {
230    for (i, stub) in ISR_STUBS.iter().enumerate() {
231        let handler_addr = *stub as usize;
232        let ist = if i == 8 {
233            // Vector 8 (Double Fault) must use dedicated IST1 stack
234            DOUBLE_FAULT_IST_INDEX
235        } else {
236            0
237        };
238
239        let flags = if i == 3 {
240            // { la: "Breakpoint" }
241            IDT_FLAG_USER_TRAP_GATE
242        } else {
243            IDT_FLAG_INTERRUPT_GATE
244        };
245
246        idt.set_handler(i, handler_addr, ist, flags);
247    }
248}

252/// Central exception dispatcher called from `common_isr_stub`.
253#[unsafe(no_mangle)]
254pub extern "C" fn exception_dispatcher(ctx: &mut ExceptionContext) {
255    match ctx.vector {
256        0 => handle_divide_error(ctx),
257        1 => handle_debug(ctx),
258        2 => handle_nmi(ctx),
259        3 => handle_breakpoint(ctx),
260        4 => handle_overflow(ctx),
261        5 => handle_bound_range(ctx),
262        6 => handle_invalid_opcode(ctx),
263        7 => handle_device_not_available(ctx),
264        8 => handle_double_fault(ctx),
265        10 => handle_invalid_tss(ctx),
266        11 => handle_segment_not_present(ctx),
267        12 => handle_stack_segment_fault(ctx),
268        13 => handle_general_protection_fault(ctx),
269        14 => handle_page_fault(ctx),
270        16 => handle_x87_floating_point(ctx),
271        17 => handle_alignment_check(ctx),
272        18 => handle_machine_check(ctx),
273        19 => handle_simd_floating_point(ctx),
274        _ => handle_unhandled_exception(ctx),
275    }
276}

360fn handle_general_protection_fault(ctx: &ExceptionContext) {
361    print_exception_banner("GENERAL PROTECTION FAULT (#GP)");
362    let mut writer = SerialPort::com1();
363    let _ = writeln!(writer, "  Error Code: {:#010x}", ctx.error_code);
364    if ctx.error_code != 0 {
365        let external = (ctx.error_code & 1) != 0;
366        let idt = ((ctx.error_code >> 1) & 1) != 0;
367        let table = match (ctx.error_code >> 2) & 1 {
368            0 => "GDT",
369            _ => "LDT",
370        };
371        let selector = (ctx.error_code >> 3) & 0x1FFF;
372        let _ = writeln!(
373            writer,
374            "  Decoded: External={}, Table={} (IDT={}), Selector Index={:#x}",
375            external, table, idt, selector
376        );
377    }
378    print_register_dump(ctx);
379    halt_system();
380}

382fn handle_page_fault(ctx: &ExceptionContext) {
383    print_exception_banner("PAGE FAULT (#PF)");
384
385    // Read faulting virtual address from CR2 register
386    let cr2: u64;
387    unsafe {
388        core::arch::asm!("mov {0}, cr2", out(reg) cr2, options(nomem, nostack, preserves_flags));
389    }
390
391    let mut writer = SerialPort::com1();
392    let _ = writeln!(writer, "  Faulting Virtual Address (CR2): {:#018x}", cr2);
393    let _ = writeln!(writer, "  Error Code:                     {:#010x}", ctx.error_code);
394
395    // Decode Page Fault Error Code:
396    let present = (ctx.error_code & (1 << 0)) != 0;
397    let write = (ctx.error_code & (1 << 1)) != 0;
398    let user = (ctx.error_code & (1 << 2)) != 0;
399    let reserved_write = (ctx.error_code & (1 << 3)) != 0;
400    let instruction_fetch = (ctx.error_code & (1 << 4)) != 0;
401    let protection_key = (ctx.error_code & (1 << 5)) != 0;
402    let shadow_stack = (ctx.error_code & (1 << 6)) != 0;
403
404    let _ = writeln!(
405        writer,
406        "  Violation Cause: [ {} | {} | {} | {} | {} ]",
407        if present { "Protection Violation" } else { "Page Not Present" },
408        if write { "Write Access" } else { "Read Access" },
409        if user { "User Mode" } else { "Kernel/Supervisor Mode" },
410        if instruction_fetch { "Instruction Fetch (NX)" } else { "Data Access" },
411        if reserved_write { "Reserved Bit Set" } else { "No Reserved Bit" }
412    );
413
414    if protection_key {
415        let _ = writeln!(writer, "  Protection Key Violation: Yes");
416    }
417    if shadow_stack {
418        let _ = writeln!(writer, "  Shadow Stack Violation:   Yes");
419    }
420
421    print_register_dump(ctx);
422    halt_system();
423}

457fn print_exception_banner(name: &str) {
458    let mut writer = SerialPort::com1();
459    let _ = writeln!(writer, "");
460    let _ = writeln!(writer, "╔═══════════════════════════════════════════════════════════════════════╗");
461    let _ = writeln!(writer, "║  FATAL CPU EXCEPTION: {:<47} ║", name);
462    let _ = writeln!(writer, "╚═══════════════════════════════════════════════════════════════════════╝");
463}

465fn print_register_dump(ctx: &ExceptionContext) {
466    let mut writer = SerialPort::com1();
467    let _ = writeln!(writer, "  Vector:     {:#04x} ({})", ctx.vector, ctx.vector);
468    let _ = writeln!(writer, "  Error Code: {:#018x}", ctx.error_code);
469    let _ = writeln!(writer, "  RIP:        {:#018x}  CS:  {:#06x}", ctx.rip, ctx.cs);
470    let _ = writeln!(writer, "  RSP:        {:#018x}  SS:  {:#06x}", ctx.rsp, ctx.ss);
471    let _ = writeln!(writer, "  RFLAGS:     {:#018x}", ctx.rflags);
472    let _ = writeln!(writer, "");
473    let _ = writeln!(writer, "  Registers:");
474    let _ = writeln!(writer, "    RAX: {:#018x}  RBX: {:#018x}", ctx.rax, ctx.rbx);
475    let _ = writeln!(writer, "    RCX: {:#018x}  RDX: {:#018x}", ctx.rcx, ctx.rdx);
476    let _ = writeln!(writer, "    RSI: {:#018x}  RDI: {:#018x}", ctx.rsi, ctx.rdi);
477    let _ = writeln!(writer, "    RBP: {:#018x}  R8:  {:#018x}", ctx.rbp, ctx.r8);
478    let _ = writeln!(writer, "    R9:  {:#018x}  R10: {:#018x}", ctx.r9, ctx.r10);
479    let _ = writeln!(writer, "    R11: {:#018x}  R12: {:#018x}", ctx.r11, ctx.r12);
480    let _ = writeln!(writer, "    R13: {:#018x}  R14: {:#018x}", ctx.r13, ctx.r14);
481    let _ = writeln!(writer, "    R15: {:#018x}", ctx.r15);
482    let _ = writeln!(writer, "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━");
483}

485fn halt_system() -> ! {
486    let mut writer = SerialPort::com1();
487    let _ = writeln!(writer, "  System halted due to fatal exception.");
488    let _ = writeln!(writer, "");
489    loop {
490        unsafe {
491            core::arch::asm!("cli");
492            core::arch::asm!("hlt");
493        }
494    }
