1// VikingOS Kernel — Interrupt Descriptor Table (IDT)
2//
3// The IDT defines how the x86-64 CPU responds to hardware interrupts,
4// processor exceptions, and software interrupts. It contains 256 gate
5// descriptors that map vector numbers (0..255) to handler routines.
6
7use core::mem::size_of;
8
9/// Total number of entries in the x86-64 IDT
10pub const IDT_ENTRIES: usize = 256;
11
12/// Gate type flag for 64-bit Interrupt Gate (Ring 0, interrupts disabled on entry)
13pub const IDT_FLAG_INTERRUPT_GATE: u8 = 0x8E;
14
15/// Gate type flag for 64-bit User Trap Gate (Ring 3 callable, e.g., Breakpoint int3)
16pub const IDT_FLAG_USER_TRAP_GATE: u8 = 0xEE;
17
18/// 64-bit IDT Gate Descriptor (16 bytes)
19#[derive(Copy, Clone)]
20#[repr(C, packed)]
21pub struct IdtEntry {
22    pub offset_low: u16,        // bits 0..15 of target address
23    pub selector: u16,          // code segment selector (0x08)
24    pub ist: u8,                // bits 0..2: IST stack index (0=none, 1..7=IST1..7)
25    pub flags: u8,              // gate type, DPL, and present flag
26    pub offset_mid: u16,        // bits 16..31 of target address
27    pub offset_high: u32,       // bits 32..63 of target address
28    pub reserved: u32,          // must be zero
29}
30
31impl IdtEntry {
32    /// Create an empty/unpresent IDT entry
33    pub const NULL: Self = Self {
34        offset_low: 0,
35        selector: 0,
36        ist: 0,
37        flags: 0,
38        offset_mid: 0,
39        offset_high: 0,
40        reserved: 0,
41    };
42
43    /// Create an active IDT gate descriptor pointing to a 64-bit handler address
44    pub fn new(handler: usize, selector: u16, ist: u8, flags: u8) -> Self {
45        Self {
46            offset_low: (handler & 0xFFFF) as u16,
47            selector,
48            ist: ist & 0x07,
49            flags,
50            offset_mid: ((handler >> 16) & 0xFFFF) as u16,
51            offset_high: ((handler >> 32) & 0xFFFFFFFF) as u32,
52            reserved: 0,
53        }
54    }
55}
56
57/// The 256-entry Interrupt Descriptor Table
58#[repr(C, align(16))]
59pub struct InterruptDescriptorTable {
60    pub entries: [IdtEntry; IDT_ENTRIES],
61}
62
63impl InterruptDescriptorTable {
64    /// Create an uninitialized IDT
65    pub const fn new() -> Self {
66        Self {
67            entries: [IdtEntry::NULL; IDT_ENTRIES],
68        }
69    }
70
71    /// Register an interrupt or exception handler for a specific vector
72    pub fn set_handler(&mut self, index: usize, handler: usize, ist: u8, flags: u8) {
73        if index < IDT_ENTRIES {
74            self.entries[index] = IdtEntry::new(
75                handler,
76                super::gdt::KERNEL_CODE_SELECTOR,
77                ist,
78                flags,
79            );
80        }
81    }
82
83    /// Load this IDT into the CPU using the `lidt` assembly instruction
84    pub fn load(&'static self) {
85        #[repr(C, packed)]
86        struct IdtPointer {
87            limit: u16,
88            base: u64,
89        }
90
91        let ptr = IdtPointer {
92            limit: (size_of::<[IdtEntry; IDT_ENTRIES]>() - 1) as u16,
93            base: self.entries.as_ptr() as u64,
94        };
95
96        unsafe {
97            core::arch::asm!(
98                "lidt [{0}]",
99                in(reg) &ptr,
100                options(readonly, nostack, preserves_flags),
101            );
102        }
103    }
104}
105
106/// Kernel singleton IDT
107static mut IDT: InterruptDescriptorTable = InterruptDescriptorTable::new();
108
109/// Return a mutable reference to the kernel IDT
110pub fn get_idt() -> &'static mut InterruptDescriptorTable {
111    unsafe { &mut *core::ptr::addr_of_mut!(IDT) }
