1// VikingOS Kernel — Hardware Interrupt Dispatcher
2//
3// This module handles the transition from low-level interrupt trampolines
4// to high-level Rust handler logic.
5
6use crate::arch::apic::LAPIC;
7use crate::kprintln;
8use crate::sync::scheduler::{SCHEDULER, Scheduler};
9use crate::drivers::keyboard::Keyboard;
10
11/// Hardware Vector Mapping
12pub const IRQ_TIMER: u8 = 32;
13pub const IRQ_KEYBOARD: u8 = 33;
14
15/// The main entry point for all hardware interrupts (vectors 32..255).
16///
17/// This function is called by the assembly trampolines after they have
18/// saved the CPU state.
19pub extern "C" fn rust_hardware_interrupt_dispatcher(vector: u8) {
20    match vector {
21        IRQ_TIMER => handle_timer_interrupt(),
22        IRQ_KEYBOARD => {
23            Keyboard::handle_interrupt();
24        },
25        _ => {
26            kprintln!("[Interrupt] Unhandled hardware interrupt: vector {}", vector);
27        }
28    }
29
30    // Signal End of Interrupt (EOI) to the LAPIC
31    LAPIC.send_eoi();
32}
33
34/// Handler for the LAPIC timer interrupt.
35fn handle_timer_interrupt() {
36    static mut TICKS: u64 = 0;
37    unsafe {
38        TICKS += 1;
39        // Print a tick every 100 ticks to avoid flooding the serial port
40        if TICKS % 100 == 0 {
41            kprintln!("[Timer] Tick count: {}", TICKS);
42        }
43    }
44
45    // TRIGGER PREEMPTIVE SCHEDULING
46    // Every timer tick, we check if it's time to switch threads.
47    let mut sched_lock = SCHEDULER.lock();
48    if let Some(ref mut sched) = *sched_lock {
49        sched.schedule();
50    }
51}
