1// VikingOS Kernel — I/O APIC Driver
2//
3// The I/O APIC manages external hardware interrupts and routes them
4// to one or more Local APICs (CPUs).
5
6#![allow(dead_code)]
7
8/// Default I/O APIC base address.
9/// In a full ACPI implementation, this would be discovered via the MADT table.
10pub const IOAPIC_BASE: usize = 0xFEE00000;
11
12// --- I/O APIC Register Offsets ---
13const REG_ID: usize = 0x00;
14const REG_WINDOW: usize = 0x10;
15
16/// Redirection Table entries start at 0x10.
17/// Each entry is 64 bits (two 32-bit registers).
18const REDIR_TABLE_START: usize = 0x10;
19
20pub struct IoApic {
21    base: usize,
22}
23
24impl IoApic {
25    /// Create a new I/O APIC handle.
26    pub const fn new() -> Self {
27        Self {
28            base: IOAPIC_BASE,
29        }
30    }
31
32    /// Read a 32-bit value from the I/O APIC.
33    /// The I/O APIC uses an index register and a data window.
34    pub unsafe fn read(&self, index: u32) -> u32 {
35        core::ptr::write_volatile((self.base + REG_ID) as *mut u32, index);
36        core::ptr::read_volatile((self.base + REG_WINDOW) as *const u32)
37    }
38
39    /// Write a 32-bit value to the I/O APIC.
40    pub unsafe fn write(&self, index: u32, value: u32) {
41        core::ptr::write_volatile((self.base + REG_ID) as *mut u32, index);
42        core::ptr::write_volatile((self.base + REG_WINDOW) as *mut u32, value);
43    }
44
45    /// Maps a hardware IRQ to a LAPIC vector.
46    ///
47    /// # Arguments
48    /// * `irq` - The hardware interrupt line (e.g., 1 for keyboard).
49    /// * `vector` - The LAPIC vector to trigger (e.g., 33).
50    /// * `dest_cpu` - The CPU to target (usually 0 for single-core).
51    pub fn map_irq(&self, irq: u8, vector: u8, dest_cpu: u8) {
52        unsafe {
53            // Redirection table entry is two 32-bit registers.
54            let low_reg = REDIR_TABLE_START + (irq as usize * 2);
55            let high_reg = low_reg + 1;
56
57            // Low 32 bits:
58            // Bits 0-7: Vector
59            // Bit 8: Mask (0 = enabled, 1 = masked)
60            // Bits 9-15: Delivery mode (000 = Fixed)
61            let low_val = vector as u32;
62            self.write(low_reg as u32, low_val);
63
64            // High 32 bits:
65            // Bits 0-7: Destination CPU
66            let high_val = dest_cpu as u32;
67            self.write(high_reg as u32, high_val);
68        }
69    }
70}
71
72/// Global I/O APIC instance.
73pub static IOAPIC: IoApic = IoApic::new();
