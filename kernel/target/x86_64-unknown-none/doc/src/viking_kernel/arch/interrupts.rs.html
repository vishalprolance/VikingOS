1// VikingOS Kernel — CPU Interrupt Control
2//
3// Provides primitives for enabling and disabling hardware interrupts.
4
5#![allow(dead_code)]
6
7/// Enable hardware interrupts (sti).
8#[inline(always)]
9pub fn enable() {
10    unsafe {
11        core::arch::asm!("sti", options(nomem, nostack, preserves_flags));
12    }
13}
14
15/// Disable hardware interrupts (cli).
16#[inline(always)]
17pub fn disable() {
18    unsafe {
19        core::arch::asm!("cli", options(nomem, nostack, preserves_flags));
20    }
21}
22
23/// Disables interrupts and returns the previous state.
24/// Returns `true` if interrupts were enabled before the call.
25#[inline(always)]
26pub fn save_and_disable() -> bool {
27    let flags: u64;
28    unsafe {
29        core::arch::asm!(
30            "pushfq",
31            "pop {0}",
32            out(reg) flags,
33            options(preserves_flags),
34            options(nomem, nostack),
35        );
36    }
37    // Bit 9 of RFLAGS is the Interrupt Flag (IF)
38    (flags & (1 << 9)) != 0
39}
40
41/// Restores the interrupt state based on the provided boolean.
42#[inline(always)]
43pub fn restore(enabled: bool) {
44    if enabled {
45        enable();
46    } else {
47        disable();
48    }
