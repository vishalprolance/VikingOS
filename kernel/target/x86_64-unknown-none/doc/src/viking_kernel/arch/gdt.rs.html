1// VikingOS Kernel — Global Descriptor Table (GDT) & Task State Segment (TSS)
2//
3// In x86-64 long mode, memory segmentation is largely disabled, but the GDT
4// is still strictly required to define:
5//   1. Privilege levels (Ring 0 for kernel, Ring 3 for userspace)
6//   2. 64-bit code segments (Long mode flag L=1, Default size D=0)
7//   3. Task State Segment (TSS) containing the Interrupt Stack Table (IST)
8//
9// The IST allows critical exceptions (such as Double Fault) to switch to a
10// known-good, dedicated stack, preventing hardware triple faults and resets.
11
12use core::mem::size_of;
13
14/// Null Segment Selector (offset 0x00)
15#[allow(dead_code)]
16pub const NULL_SELECTOR: u16 = 0x00;
17/// Kernel Code Segment Selector (Index 1, RPL 0, offset 0x08)
18pub const KERNEL_CODE_SELECTOR: u16 = 0x08;
19/// Kernel Data Segment Selector (Index 2, RPL 0, offset 0x10)
20pub const KERNEL_DATA_SELECTOR: u16 = 0x10;
21/// User Data Segment Selector (Index 3, RPL 3, offset 0x18 | 3 = 0x1B)
22#[allow(dead_code)]
23pub const USER_DATA_SELECTOR: u16 = 0x18 | 3;
24/// User Code Segment Selector (Index 4, RPL 3, offset 0x20 | 3 = 0x23)
25#[allow(dead_code)]
26pub const USER_CODE_SELECTOR: u16 = 0x20 | 3;
27/// Task State Segment Selector (Index 5, RPL 0, offset 0x28)
28pub const TSS_SELECTOR: u16 = 0x28;
29
30/// Dedicated stack size for Double Fault handler (16 KiB)
31pub const DOUBLE_FAULT_STACK_SIZE: usize = 16384;
32
33/// Index in TSS IST for Double Fault handler
34pub const DOUBLE_FAULT_IST_INDEX: u8 = 1;

36/// Task State Segment (64-bit x86 architecture)
37///
38/// Holds Privilege Level 0..2 stack pointers (RSP0-RSP2) and seven
39/// Interrupt Stack Table pointers (IST1-IST7).
40#[repr(C, packed)]
41pub struct TaskStateSegment {
42    reserved_0: u32,
43    pub rsp0: u64,
44    pub rsp1: u64,
45    pub rsp2: u64,
46    reserved_1: u64,
47    pub ist1: u64,
48    pub ist2: u64,
49    pub ist3: u64,
50    pub ist4: u64,
51    pub ist5: u64,
52    pub ist6: u64,
53    pub ist7: u64,
54    reserved_2: u64,
55    reserved_3: u16,
56    pub iopb_offset: u16,
57}
58
59impl TaskStateSegment {
60    pub const fn new() -> Self {
61        Self {
62            reserved_0: 0,
63            rsp0: 0,
64            rsp1: 0,
65            rsp2: 0,
66            reserved_1: 0,
67            ist1: 0,
68            ist2: 0,
69            ist3: 0,
70            ist4: 0,
71            ist5: 0,
72            ist6: 0,
73            ist7: 0,
74            reserved_2: 0,
75            reserved_3: 0,
76            // Setting iopb_offset >= sizeof(TSS) disables I/O permission bitmap
77            iopb_offset: 104,
78        }
79    }
80}

82/// 16-byte aligned stack for Double Fault exception handling
83#[repr(align(16))]
84#[allow(dead_code)]
85struct DoubleFaultStack([u8; DOUBLE_FAULT_STACK_SIZE]);
86
87/// Dedicated stack memory for Double Faults
88static mut DOUBLE_FAULT_STACK: DoubleFaultStack = DoubleFaultStack([0; DOUBLE_FAULT_STACK_SIZE]);
89
90/// Kernel Task State Segment singleton
91static mut TSS: TaskStateSegment = TaskStateSegment::new();
92
93/// Global Descriptor Table entries:
94/// 0: Null descriptor
95/// 1: Kernel Code 64-bit
96/// 2: Kernel Data 64-bit
97/// 3: User Data 64-bit
98/// 4: User Code 64-bit
99/// 5: TSS Descriptor (low 64 bits)
100/// 6: TSS Descriptor (high 64 bits)
101static mut GDT: [u64; 7] = [0; 7];
102
103/// Pseudo-descriptor loaded into the CPU GDTR register
104#[repr(C, packed)]
105struct GdtPointer {
106    limit: u16,
107    base: u64,
108}
109
110/// Initialize and activate the VikingOS GDT and TSS.
111///
112/// Sets up kernel and user segments, initializes the double-fault IST stack,
113/// loads the GDT using `lgdt`, reloads segment registers with new selectors,
114/// and activates the TSS using `ltr`.
115pub fn init() {
116    unsafe {
117        // 1. Configure TSS with Double Fault IST1 stack
118        let stack_start = core::ptr::addr_of!(DOUBLE_FAULT_STACK) as u64;
119        let stack_top = stack_start + (DOUBLE_FAULT_STACK_SIZE as u64);
120        TSS.ist1 = stack_top;
121
122        // 2. Build GDT entries
123        // Entry 0: Null Descriptor
124        GDT[0] = 0x0000_0000_0000_0000;
125
126        // Entry 1: Kernel Code 64-bit (offset 0x08)
127        // Access: Present (1), Ring 0 (00), Code/Data (1), Executable (1), Readable (1), Accessed (0) -> 0x9A
128        // Flags: Long mode (1), Default size (0), Granularity (0) -> 0x20
129        GDT[1] = 0x0020_9A00_0000_0000;
130
131        // Entry 2: Kernel Data 64-bit (offset 0x10)
132        // Access: Present (1), Ring 0 (00), Code/Data (1), Executable (0), Writable (1), Accessed (0) -> 0x92
133        // Flags: 0x00
134        GDT[2] = 0x0000_9200_0000_0000;
135
136        // Entry 3: User Data 64-bit (offset 0x18)
137        // Access: Present (1), Ring 3 (11), Code/Data (1), Executable (0), Writable (1), Accessed (0) -> 0xF2
138        // Flags: 0x00
139        GDT[3] = 0x0000_F200_0000_0000;
140
141        // Entry 4: User Code 64-bit (offset 0x20)
142        // Access: Present (1), Ring 3 (11), Code/Data (1), Executable (1), Readable (1), Accessed (0) -> 0xFA
143        // Flags: Long mode (1) -> 0x20
144        GDT[4] = 0x0020_FA00_0000_0000;
145
146        // Entries 5 & 6: 16-byte TSS Descriptor (offset 0x28)
147        let tss_addr = core::ptr::addr_of!(TSS) as u64;
148        let tss_limit = (size_of::<TaskStateSegment>() - 1) as u64;
149
150        let limit_low = tss_limit & 0xFFFF;
151        let base_low = (tss_addr & 0x00FFFFFF) << 16;
152        // Type 0x89: Present (bit 47), DPL 0 (bits 45-46), 64-bit TSS Available (bits 40-43: 0b1001)
153        let access = 0x89u64 << 40;
154        let limit_high = ((tss_limit >> 16) & 0x0F) << 48;
155        let base_mid = ((tss_addr >> 24) & 0xFF) << 56;
156
157        let tss_low = limit_low | base_low | access | limit_high | base_mid;
158        let tss_high = tss_addr >> 32;
159
160        GDT[5] = tss_low;
161        GDT[6] = tss_high;
162
163        // 3. Load GDT into CPU
164        let gdt_ptr = GdtPointer {
165            limit: (size_of::<[u64; 7]>() - 1) as u16,
166            base: core::ptr::addr_of!(GDT) as u64,
167        };
168
169        core::arch::asm!(
170            "lgdt [{0}]",
171            in(reg) &gdt_ptr,
172            options(readonly, nostack, preserves_flags),
173        );
174
175        // 4. Reload CS via far return (retfq) and data segment registers
176        core::arch::asm!(
177            "push {code_sel}",
178            "lea {tmp}, [2f + rip]",
179            "push {tmp}",
180            "retfq",
181            "2:",
182            "mov ds, {data_sel:x}",
183            "mov es, {data_sel:x}",
184            "mov fs, {data_sel:x}",
185            "mov gs, {data_sel:x}",
186            "mov ss, {data_sel:x}",
187            code_sel = in(reg) (KERNEL_CODE_SELECTOR as u64),
188            data_sel = in(reg) KERNEL_DATA_SELECTOR,
189            tmp = out(reg) _,
190        );
191
192        // 5. Load Task Register (LTR)
193        core::arch::asm!(
194            "ltr {0:x}",
195            in(reg) TSS_SELECTOR,
196            options(nostack, preserves_flags),
197        );
198    }
