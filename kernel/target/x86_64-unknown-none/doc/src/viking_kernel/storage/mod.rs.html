1// VikingOS Kernel — Storage Abstractions
2//
3// This module defines the common interfaces for block devices,
4// decoupling the VFS from specific hardware drivers.
5
6pub enum StorageError {
7    ReadError,
8    WriteError,
9    InvalidBlock,
10    DeviceNotReady,
11}
12
13/// Trait for a block-based storage device.
14pub trait BlockDevice: Send + Sync {
15    /// Read a single block of data into the provided buffer.
16    fn read_block(&self, block_id: u64, buffer: &mut [u8]) -> Result<(), StorageError>;
17
18    /// Write a single block of data from the provided buffer.
19    fn write_block(&self, block_id: u64, buffer: &[u8]) -> Result<(), StorageError>;
20
21    /// Returns the size of a single block in bytes (usually 512).
22    fn block_size(&self) -> usize;
23
24    /// Returns the total number of blocks available on the device.
25    fn total_blocks(&self) -> u64;
26}
