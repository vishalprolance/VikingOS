1// VikingOS Kernel — Entry Point
2//
3// This is the first Rust code that executes after the Limine bootloader
4// hands control to our kernel. It initializes serial output, prints the
5// boot banner, reports diagnostic information, and halts.
6//
7// Target: x86_64-unknown-none (bare metal, no_std)
8// Bootloader: Limine (boot protocol revision 3+)
9

11#![no_main]

36// ── Limine Protocol Requests ─────────────────────────────────────────
37//
38// These static structures are placed in special ELF sections so the
39// Limine bootloader can discover them before transferring control.
40// The bootloader fills in their responses before calling _start().
41
42/// Marks the beginning of the Limine requests region.
43#[used]
44#[unsafe(link_section = ".requests_start_marker")]
45static _START_MARKER: RequestsStartMarker = RequestsStartMarker::new();
46
47/// Marks the end of the Limine requests region.
48#[used]
49#[unsafe(link_section = ".requests_end_marker")]
50static _END_MARKER: RequestsEndMarker = RequestsEndMarker::new();
51
52/// Base revision — ensures bootloader/kernel protocol compatibility.
53#[used]
54#[unsafe(link_section = ".requests")]
55static BASE_REVISION: BaseRevision = BaseRevision::with_revision(3);
56
57/// Request bootloader name and version.
58#[used]
59#[unsafe(link_section = ".requests")]
60static BOOTLOADER_INFO: BootloaderInfoRequest = BootloaderInfoRequest::new();
61
62/// Request the physical memory map.
63#[used]
64#[unsafe(link_section = ".requests")]
65static MEMORY_MAP: MemmapRequest = MemmapRequest::new();
66
67/// Request the Higher Half Direct Map offset.
68#[used]
69#[unsafe(link_section = ".requests")]
70static HHDM: HhdmRequest = HhdmRequest::new();
71
72/// Request a linear framebuffer.
73#[used]
74#[unsafe(link_section = ".requests")]
75static FRAMEBUFFER: FramebufferRequest = FramebufferRequest::new();

93// ── Kernel Entry Point ───────────────────────────────────────────────
94
95/// Kernel entry point — called by Limine after boot.
96///
97/// Initializes serial output, prints the boot banner and diagnostics,
98/// then halts the CPU. This function never returns.
99#[unsafe(no_mangle)]
100extern "C" fn _start() -> ! {
101    // Initialize COM1 serial port for diagnostic output
102    serial::init();
103
104    // We need a mutable reference for fmt::Write
105    let mut writer = serial::SerialPort::com1();
106
107    // ── Boot Banner ──────────────────────────────────────────────
108    let _ = writeln!(writer, "{}", BANNER);
109    let _ = writeln!(writer, "  VikingOS Kernel v{}", KERNEL_VERSION);
110    let _ = writeln!(writer, "  Architecture: {}", KERNEL_ARCH);
111    let _ = writeln!(writer, "  License: MIT");
112    let _ = writeln!(writer, "");
113
114    // ── Protocol Check ───────────────────────────────────────────
115    if !BASE_REVISION.is_supported() {
116        let _ = writeln!(writer, "[FATAL] Limine base revision not supported!");
117        halt_loop();
118    }
119    let _ = writeln!(writer, "[  OK  ] Limine base revision supported");
120
121    // ── Bootloader Info ──────────────────────────────────────────
122    if let Some(info) = BOOTLOADER_INFO.response() {
123        let name = info.name();
124        let version = info.version();
125        let _ = writeln!(writer, "[  OK  ] Bootloader: {} v{}", name, version);
126    } else {
127        let _ = writeln!(writer, "[ WARN ] Bootloader info not available");
128    }
129
130    // ── HHDM ─────────────────────────────────────────────────────
131    if let Some(hhdm) = HHDM.response() {
132        let _ = writeln!(
133            writer,
134            "[  OK  ] HHDM offset: {:#018x}",
135            hhdm.offset
136        );
137    } else {
138        let _ = writeln!(writer, "[ WARN ] HHDM not available");
139    }
140
141    // ── Memory Map ───────────────────────────────────────────────
142    if let Some(memmap) = MEMORY_MAP.response() {
143        let entries = memmap.entries();
144        let _ = writeln!(writer, "[  OK  ] Memory map: {} entries", entries.len());
145        let _ = writeln!(writer, "");
146        let _ = writeln!(
147            writer,
148            "         {:>18}  {:>18}  {:>12}  {}",
149            "Base", "Length", "Size", "Type"
150        );
151        let _ = writeln!(
152            writer,
153            "         {:->18}  {:->18}  {:->12}  {:->20}",
154            "", "", "", ""
155        );
156
157        let mut total_usable: u64 = 0;
158        let mut total_all: u64 = 0;
159
160        for entry in entries.iter() {
161            let base = entry.base;
162            let length = entry.length;
163            let entry_type = entry.type_;
164
165            total_all += length;
166            if entry_type == limine::memmap::MEMMAP_USABLE {
167                total_usable += length;
168            }
169
170            let type_name = match entry_type {
171                limine::memmap::MEMMAP_USABLE => "Usable",
172                limine::memmap::MEMMAP_RESERVED => "Reserved",
173                limine::memmap::MEMMAP_ACPI_RECLAIMABLE => "ACPI Reclaimable",
174                limine::memmap::MEMMAP_ACPI_NVS => "ACPI NVS",
175                limine::memmap::MEMMAP_BAD_MEMORY => "Bad Memory",
176                limine::memmap::MEMMAP_BOOTLOADER_RECLAIMABLE => "Bootloader Recl.",
177                limine::memmap::MEMMAP_EXECUTABLE_AND_MODULES => "Kernel+Modules",
178                limine::memmap::MEMMAP_FRAMEBUFFER => "Framebuffer",
179                _ => "Unknown",
180            };
181
182            let _ = writeln!(
183                writer,
184                "         {:#018x}  {:#018x}  {:>9} KB  {}",
185                base,
186                length,
187                length / 1024,
188                type_name,
189            );
190        }
191
192        let _ = writeln!(writer, "");
193        let _ = writeln!(
194            writer,
195            "         Total memory:  {} MB ({} KB)",
196            total_all / (1024 * 1024),
197            total_all / 1024
198        );
199        let _ = writeln!(
200            writer,
201            "         Usable memory: {} MB ({} KB)",
202            total_usable / (1024 * 1024),
203            total_usable / 1024
204        );
205    } else {
206        let _ = writeln!(writer, "[ FAIL ] Memory map not available!");
207    }
208
209    // ── Framebuffer ──────────────────────────────────────────────
210    if let Some(fb_response) = FRAMEBUFFER.response() {
211        let fbs = fb_response.framebuffers();
212        if let Some(fb) = fbs.first() {
213            let _ = writeln!(
214                writer,
215                "[  OK  ] Framebuffer: {}x{} @ {} bpp, pitch {}",
216                fb.width,
217                fb.height,
218                fb.bpp,
219                fb.pitch,
220            );
221        } else {
222            let _ = writeln!(writer, "[ WARN ] No framebuffer available");
223        }
224    } else {
225        let _ = writeln!(writer, "[ WARN ] Framebuffer request not fulfilled");
226    }
227
228    // ── CPU Architecture: GDT, TSS & IDT (Milestone 2) ───────────
229    let _ = writeln!(writer, "");
230    let _ = writeln!(writer, "[ INIT ] Initializing x86-64 CPU architecture...");
231    arch::init();
232    let _ = writeln!(writer, "[  OK  ] GDT loaded (CS: 0x08, DS: 0x10, User CS: 0x23, User DS: 0x1B)");
233    let _ = writeln!(writer, "[  OK  ] TSS active (Selector: 0x28, IST1 Double Fault stack: 16 KiB)");
234    let _ = writeln!(writer, "[  OK  ] IDT loaded (256 vectors, 32 exception handlers registered)");
235
236    // ── Memory Management Subsystem (Milestone 3) ────────────────
237    kprintln!();
238    kprintln!("[ INIT ] Initializing Memory Management Subsystem (Milestone 3)...");
239
240    let hhdm_offset = HHDM
241        .response()
242        .expect("Limine HHDM response required for memory subsystem")
243        .offset;
244    let memmap_entries = MEMORY_MAP
245        .response()
246        .expect("Limine Memory Map response required for memory subsystem")
247        .entries();
248
249    memory::init(hhdm_offset, memmap_entries);
250
251    // ── PMM Verification ─────────────────────────────────────────
252    kprintln!("[ TEST ] Verifying Physical Memory Manager (PMM)...");
253    let test_frame = memory::pmm::allocate_frame().expect("PMM: Failed to allocate test frame");
254    kprintln!("[  OK  ] Allocated physical frame at {:#018x}", test_frame.as_u64());
255
256    // Write test signature to allocated frame via HHDM
257    unsafe {
258        let ptr = memory::phys_to_virt(test_frame).as_mut_ptr::<u64>();
259        *ptr = 0xDEADC0DE_CAFEBABE;
260        assert_eq!(*ptr, 0xDEADC0DE_CAFEBABE, "PMM: Memory readback mismatch");
261    }
262    kprintln!("[  OK  ] Frame read/write verification passed via HHDM");
263
264    // Test multi-frame allocation
265    let multi_frames = memory::pmm::allocate_frames(4).expect("PMM: Failed to allocate 4 contiguous frames");
266    kprintln!("[  OK  ] Allocated 4 contiguous frames at {:#018x}", multi_frames.as_u64());
267
268    // Deallocate test frames
269    memory::pmm::deallocate_frame(test_frame);
270    memory::pmm::deallocate_frames(multi_frames, 4);
271    kprintln!("[  OK  ] Deallocated test frames back to PMM frame pool");
272
273    // ── VMM Verification ─────────────────────────────────────────
274    kprintln!("[ TEST ] Verifying Virtual Memory Manager (VMM 4-Level Paging)...");
275    let test_virt = memory::VirtAddr::new(0xFFFF_C000_0000_0000);
276    let vmm_frame = memory::pmm::allocate_zeroed_frame().expect("VMM test: frame allocation failed");
277
278    memory::vmm::map_page(
279        test_virt,
280        vmm_frame,
281        memory::PageFlags::PRESENT | memory::PageFlags::WRITABLE | memory::PageFlags::NO_EXECUTE,
282    ).expect("VMM: Failed to map test page");
283
284    let translated = memory::vmm::translate(test_virt).expect("VMM: Translation failed for mapped page");
285    assert_eq!(translated, vmm_frame, "VMM: Translation physical address mismatch");
286    kprintln!("[  OK  ] Page mapped: {:#018x} -> {:#018x} (translated successfully)", test_virt.as_u64(), translated.as_u64());
287
288    // Write through mapped virtual address
289    unsafe {
290        let ptr = test_virt.as_mut_ptr::<u64>();
291        *ptr = 0x1234_5678_9ABC_DEF0;
292        assert_eq!(*ptr, 0x1234_5678_9ABC_DEF0, "VMM: Virtual page read/write mismatch");
293    }
294    kprintln!("[  OK  ] Virtual page read/write verified");
295
296    // Unmap test page
297    let unmapped_phys = memory::vmm::unmap_page(test_virt).expect("VMM: Failed to unmap test page");
298    assert_eq!(unmapped_phys, vmm_frame);
299    assert!(memory::vmm::translate(test_virt).is_none(), "VMM: Page still translated after unmap");
300    memory::pmm::deallocate_frame(vmm_frame);
301    kprintln!("[  OK  ] Virtual page unmapped and TLB invalidated successfully");
302
303    // ── Kernel Heap & Alloc Collections Verification ─────────────
304    kprintln!("[ TEST ] Verifying Kernel Dynamic Heap (`alloc` crate)...");
305
306    // 1. Box test
307    let boxed_val = alloc::boxed::Box::new(0x42_u64);
308    assert_eq!(*boxed_val, 0x42);
309    kprintln!("[  OK  ] Box<u64> allocated at {:p} with value {}", boxed_val, *boxed_val);
310
311    // 2. Vec test with reallocation
312    let mut vec: alloc::vec::Vec<u64> = alloc::vec::Vec::new();
313    for i in 0..256 {
314        vec.push(i * 10);
315    }
316    assert_eq!(vec.len(), 256);
317    assert_eq!(vec[100], 1000);
318    kprintln!("[  OK  ] Vec<u64> populated (len: {}, capacity: {}, vec[100] == {})", vec.len(), vec.capacity(), vec[100]);
319
320    // 3. String & format! test
321    let formatted = alloc::format!("VikingOS Kernel v{} — Heap Allocator Fully Operational!", KERNEL_VERSION);
322    kprintln!("[  OK  ] String & format!: \"{}\"", formatted);
323
324    // 4. Arc reference-counting test
325    let arc_val = alloc::sync::Arc::new(9999_u32);
326    let arc_clone = alloc::sync::Arc::clone(&arc_val);
327    assert_eq!(alloc::sync::Arc::strong_count(&arc_val), 2);
328    drop(arc_clone);
329    assert_eq!(alloc::sync::Arc::strong_count(&arc_val), 1);
330    kprintln!("[  OK  ] Arc<u32> reference counting verified (strong count: {})", alloc::sync::Arc::strong_count(&arc_val));
331
332    // ── Memory Statistics Summary ────────────────────────────────
333    let pmm_stats = memory::pmm::get_stats();
334    let (heap_alloc, heap_cap) = memory::heap::stats();
335    kprintln!();
336    kprintln!("==================== MEMORY SUBSYSTEM STATUS ====================");
337    kprintln!("  Physical Memory:");
338    kprintln!("    Total Frames: {:>8} ({} MB)", pmm_stats.total_frames, pmm_stats.total_bytes / (1024 * 1024));
339    kprintln!("    Free Frames:  {:>8} ({} MB)", pmm_stats.free_frames, pmm_stats.free_bytes / (1024 * 1024));
340    kprintln!("    Used Frames:  {:>8} ({} MB)", pmm_stats.used_frames, pmm_stats.used_bytes / (1024 * 1024));
341    kprintln!("  Virtual Memory:");
342    kprintln!("    Active PML4:  {:#018x}", memory::vmm::read_cr3().as_u64());
343    kprintln!("    HHDM Offset:  {:#018x}", memory::hhdm_offset());
344    kprintln!("  Kernel Heap:");
345    kprintln!("    Heap Base:    {:#018x}", memory::heap::HEAP_START);
346    kprintln!("    Capacity:     {:>8} KB ({} MB)", heap_cap / 1024, heap_cap / (1024 * 1024));
347    kprintln!("    Allocated:    {:>8} bytes ({} KB)", heap_alloc, heap_alloc / 1024);
348    kprintln!("    Free:         {:>8} KB", (heap_cap - heap_alloc) / 1024);
349    kprintln!("=================================================================");
350
351    // ── Hardware Interrupts & LAPIC (Milestone 4) ──────────────
352    kprintln!("");
353    kprintln!("[ INIT ] Initializing Local APIC...");
354    arch::apic::LAPIC.init();
355    kprintln!("[  OK  ] LAPIC initialized and SVR configured");
356
357    // Register Hardware Interrupt Handlers in IDT
358    let idt = arch::idt::get_idt();
359    // Vector 32: APIC Timer
360    idt.set_handler(32, arch::exceptions::hw_isr_stub_32 as usize, 0, arch::idt::IDT_FLAG_INTERRUPT_GATE);
361    kprintln!("[  OK  ] IDT: Registered APIC Timer handler (Vector 32)");
362
363    // Vector 33: Keyboard
364    idt.set_handler(33, arch::exceptions::hw_isr_stub_33 as usize, 0, arch::idt::IDT_FLAG_INTERRUPT_GATE);
365    kprintln!("[  OK  ] IDT: Registered Keyboard handler (Vector 33)");
366
367    // Setup APIC Timer: 100Hz periodic
368    arch::apic::LAPIC.setup_timer(32, 100);
369    kprintln!("[  OK  ] APIC Timer configured to 100Hz");
370
371    // ── I/O APIC & External Interrupts (Milestone 6) ─────────────
372    kprintln!("");
373    kprintln!("[ INIT ] Initializing I/O APIC...");
374    arch::ioapic::IOAPIC.map_irq(1, 33, 0); // Map Keyboard IRQ 1 to Vector 33, CPU 0
375    kprintln!("[  OK  ] I/O APIC configured: IRQ 1 -> Vector 33");
376
377    // Initialize Keyboard Driver
378    drivers::keyboard::Keyboard::init();
379    kprintln!("[  OK  ] Keyboard driver initialized");
380
381    // ── Graphics & Text Console (Milestone 6) ────────────────────
382    kprintln!("");
383    kprintln!("[ INIT ] Initializing Graphical Console...");
384    if let Some(fb_resp) = FRAMEBUFFER.response().and_then(|r| r.framebuffers().first()) {
385        unsafe {
386            let fb = Box::leak(Box::new(drivers::framebuffer::Framebuffer::new(
387                fb_resp.address,
388                fb_resp.width,
389                fb_resp.height,
390                fb_resp.bpp,
391                fb_resp.pitch,
392            )));
393            drivers::console::init(fb);
394
395            // --- GUI Engine Demonstration ---
396            kprintln!("[ INIT ] Booting GUI Engine...");
397            let mut canvas = drivers::gui::Canvas::new(fb);
398            canvas.clear(0x008080); // Teal background
399
400            let mut wm = drivers::gui::WindowManager::new();
401            wm.add_window(drivers::gui::Window::new("Welcome", 100, 100, 400, 300, 0xCCCCCC));
402            wm.add_window(drivers::gui::Window::new("System Info", 200, 200, 300, 200, 0xDDDDDD));
403
404            let taskbar = drivers::gui::Taskbar::new();
405
406            // Render the desktop
407            wm.render(&mut canvas);
408            taskbar.draw(&mut canvas, fb_resp.width, fb_resp.height);
409
410            kprintln!("[  OK  ] GUI Desktop rendered to framebuffer");
411        }
412        kprintln!("[  OK  ] Console initialized ({}x{}, {}bpp)", fb_resp.width, fb_resp.height, fb_resp.bpp);
413    } else {
414        kprintln!("[ FAIL ] Framebuffer not found!");
415    }
416
417    // ── Storage & VFS (Milestone 8) ─────────────────────────────
418    kprintln!("");
419    kprintln!("[ INIT ] Initializing Storage Subsystem...");
420    // In a real boot, we'd use VirtioBlock here. For now, we bootstrap with Ramdisk.
421    let ramdisk_data = b"Hello from the VikingOS Ramdisk! This is a user-mode binary simulation.";
422    let ramdisk = Box::leak(Box::new(crate::fs::ramdisk::Ramdisk::new(ramdisk_data)));
423    kprintln!("[  OK  ] Ramdisk filesystem mounted at /");
424
425    // ── Syscalls & User-Mode (Milestone 9) ─────────────────────
426    kprintln!("");
427    kprintln!("[ INIT ] Configuring System Call Interface...");
428    unsafe {
429        syscall::init_syscall_msrs();
430    }
431    kprintln!("[  OK  ] Syscall MSRs configured (LSTAR -> dispatcher)");
432
433    // ── Application Loading & User Processes ──────────────────
434    kprintln!("");
435    kprintln!("[ INIT ] Loading user-mode applications...");
436
437    // Embedded application binaries
438    // In a real system, these would be actual compiled ELF binaries.
439    // Here we simulate them as byte arrays.
440    static NOTEPAD_BIN: process::UserBinary = process::UserBinary {
441        name: "Notepad",
442        code: b"NOTEPAD_SIMULATED_CODE",
443        entry_point: 0x400000,
444    };
445
446    static SYS_MONITOR_BIN: process::UserBinary = process::UserBinary {
447        name: "System Monitor",
448        code: b"SYS_MONITOR_SIMULATED_CODE",
449        entry_point: 0x400000,
450    };
451
452    unsafe {
453        let notepad_proc = process::Process::load_binary(process::ProcessId(1), &NOTEPAD_BIN)
454            .expect("Failed to load Notepad");
455        kprintln!("[  OK  ] Loaded Notepad (PID 1)");
456
457        let monitor_proc = process::Process::load_binary(process::ProcessId(2), &SYS_MONITOR_BIN)
458            .expect("Failed to load System Monitor");
459        kprintln!("[  OK  ] Loaded System Monitor (PID 2)");
460    }
461
462
463    // ── Exception Handling Verification ──────────────────────────
464    let _ = writeln!(writer, "[ TEST ] Triggering CPU breakpoint trap (int3)...");
465    unsafe {
466        core::arch::asm!("int3");
467    }
468    let _ = writeln!(writer, "[  OK  ] Execution resumed successfully after breakpoint trap!");
469    let _ = writeln!(writer, "[  OK  ] CPU core architecture & exceptions fully operational");
470
471    // --- FINAL STEP: ENABLE INTERRUPTS ---
472    kprintln!("");
473    kprintln!("[ FINAL ] Enabling hardware interrupts... (The system will now start ticking)");
474    arch::interrupts::enable();
475    kprintln!("[  OK  ] Interrupts enabled. Heartbeat active.");
476
477    // ── Memory Management Subsystem (Milestone 3) ────────────────
478    kprintln!();
479    kprintln!("[ INIT ] Initializing Memory Management Subsystem (Milestone 3)...");
480
481    let hhdm_offset = HHDM
482        .response()
483        .expect("Limine HHDM response required for memory subsystem")
484        .offset;
485    let memmap_entries = MEMORY_MAP
486        .response()
487        .expect("Limine Memory Map response required for memory subsystem")
488        .entries();
489
490    memory::init(hhdm_offset, memmap_entries);
491
492    // ── PMM Verification ─────────────────────────────────────────
493    kprintln!("[ TEST ] Verifying Physical Memory Manager (PMM)...");
494    let test_frame = memory::pmm::allocate_frame().expect("PMM: Failed to allocate test frame");
495    kprintln!("[  OK  ] Allocated physical frame at {:#018x}", test_frame.as_u64());
496
497    // Write test signature to allocated frame via HHDM
498    unsafe {
499        let ptr = memory::phys_to_virt(test_frame).as_mut_ptr::<u64>();

502    }
503    kprintln!("[  OK  ] Frame read/write verification passed via HHDM");
504
505    // Test multi-frame allocation
506    let multi_frames = memory::pmm::allocate_frames(4).expect("PMM: Failed to allocate 4 contiguous frames");
507    kprintln!("[  OK  ] Allocated 4 contiguous frames at {:#018x}", multi_frames.as_u64());
508

513
514    // ── VMM Verification ─────────────────────────────────────────
515    kprintln!("[ TEST ] Verifying Virtual Memory Manager (VMM 4-Level Paging)...");
516    let test_virt = memory::VirtAddr::new(0xFFFF_C000_0000_0000);

518

526    assert_eq!(translated, vmm_frame, "VMM: Translation physical address mismatch");

542    kprintln!("[  OK  ] Virtual page unmapped and TLB invalidated successfully");

545    kprintln!("[ TEST ] Verifying Kernel Dynamic Heap (`alloc` crate)...");
546
547    // 1. Box test

629// ── Panic Handler ────────────────────────────────────────────────────
630
631/// Panic handler — prints the panic message to serial and halts.
632///
633/// This is required by `#![no_std]`. In a bare-metal kernel, a panic
634/// is unrecoverable — we print what we can and stop the CPU.
635#[panic_handler]
636fn panic(info: &PanicInfo) -> ! {
637    let mut writer = serial::SerialPort::com1();
638    // Re-initialize in case the serial port wasn't set up yet
639    writer.init();
640
641    let _ = writeln!(writer, "");
642    let _ = writeln!(writer, "!!! KERNEL PANIC !!!");
643    let _ = writeln!(writer, "{}", info);
644    let _ = writeln!(writer, "");
645    let _ = writeln!(writer, "System halted.");
646
647    halt_loop();
648}

650// ── CPU Halt ─────────────────────────────────────────────────────────
651
652/// Halt the CPU in an infinite loop.
653///
654/// Disables interrupts and executes HLT repeatedly. This is the safe
655/// way to stop execution on x86 — HLT waits for the next interrupt,
656/// and since interrupts are disabled, the CPU stops permanently.
657fn halt_loop() -> ! {
658    loop {
659        unsafe {
660            core::arch::asm!("cli"); // Disable interrupts
661            core::arch::asm!("hlt"); // Halt until next interrupt (never)
662        }
663    }
