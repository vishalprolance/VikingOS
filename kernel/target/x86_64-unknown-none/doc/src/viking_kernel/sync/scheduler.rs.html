1// VikingOS Kernel — Round-Robin Scheduler
2//
3// Manages the execution of multiple kernel threads.
4// The scheduler is triggered by the LAPIC timer to ensure preemption.
5
6use alloc::collections::VecDeque;
7use alloc::vec::Vec;
8use alloc::sync::Arc;
9use crate::sync::SpinLock;
10use crate::thread::{Thread, ThreadId, ThreadState};
11use crate::memory::pmm;
12use crate::memory::vmm;
13use crate::arch::context::switch_context;
14
15/// The global scheduler state.
16pub struct Scheduler {
17    /// The queue of threads ready to run.
18    ready_queue: VecDeque<ThreadId>,
19    /// All existing threads, indexed by their ID.
20    threads: Vec<Thread>,
21    /// The ID of the thread currently executing on the CPU.
22    current_thread: Option<ThreadId>,
23    /// Counter for assigning unique thread IDs.
24    next_id: ThreadId,
25}
26
27impl Scheduler {
28    /// Create a new, empty scheduler.
29    pub const fn new() -> Self {
30        // The Scheduler cannot be const-initialized due to heap collections.
31        // We leave this as a panic/unreachable since the global SCHEDULER
32        // is initialized dynamically via init().
33        panic!("Scheduler cannot be const-initialized")
34    }
35}
36
37/// Use a dynamic initialization pattern for the global scheduler.
38pub static SCHEDULER: SpinLock<Option<Scheduler>> = SpinLock::new(None);
39
40/// Initialize the global scheduler.
41pub fn init() {
42    let mut sched = SCHEDULER.lock();
43    *sched = Some(Scheduler {
44        ready_queue: VecDeque::new(),
45        threads: Vec::new(),
46        current_thread: None,
47        next_id: 0,
48    });
49}
50
51/// Helper to spawn a kernel thread.
52pub fn spawn_thread(f: fn()) {
53    let frame = pmm::allocate_frame().expect("Scheduler: Failed to allocate kernel stack frame");
54    let mut sched = SCHEDULER.lock();
55    if let Some(ref mut s) = *sched {
56        let pid = s.next_id;
57        s.next_id += 1;
58
59        let thread = Thread::new(pid, f as usize, frame);
60        s.threads.push(thread);
61        s.ready_queue.push_back(pid);
62    }
63}
64
65impl Scheduler {
66    /// Spawn a kernel thread.
67    pub fn spawn_thread(&mut self, f: fn()) {
68        let frame = pmm::allocate_frame().expect("Scheduler: Failed to allocate kernel stack frame");
69        let pid = self.next_id;
70        self.next_id += 1;
71
72        let thread = Thread::new(pid, f as usize, frame);
73        self.threads.push(thread);
74        self.ready_queue.push_back(pid);
75    }
76
77    /// Spawn a user process.
78    ///
79    /// # Arguments
80    /// * `entry_point` - The virtual address where the user process starts.
81    /// * `user_stack` - The virtual address of the user-mode stack.
82    /// * `cr3` - The physical address of the process's page table.
83    pub fn spawn_process(&mut self, entry_point: crate::memory::VirtAddr, user_stack: crate::memory::VirtAddr, cr3: crate::memory::PhysAddr) {
84        // 1. Allocate a kernel stack for the process
85        let frame = pmm::allocate_frame().expect("Scheduler: Failed to allocate kernel stack frame");
86
87        // 2. Create the Process object
88        let pid = self.next_id;
89        self.next_id += 1;
90
91        let mut process = unsafe {
92            crate::process::Process::new(
93                crate::process::ProcessId(pid as u64),
94                cr3,
95                frame,
96                user_stack,
97            )
98        };
99
100        // 3. Prepare the stack for the first transition to Ring 3
101        let stack_ptr = process.prepare_user_transition(entry_point);
102
103        // 4. We need to wrap this in a Thread structure for the scheduler to manage it.
104        // Since our scheduler currently only handles `Thread` objects, we treat the
105        // user process as a special kind of thread that starts in Ring 3.
106        let thread = Thread::new(pid, stack_ptr, frame);
107
108        // Note: To fully support the CR3 switch, the scheduler's `schedule`
109        // function must be updated to check if the next thread belongs to a process.
110        self.threads.push(thread);
111        self.ready_queue.push_back(pid);
112    }
113
114    /// Trigger a context switch to the next ready thread.
115    pub fn schedule(&mut self) {
116        let current_id = match self.current_thread {
117            Some(id) => id,
118            None => {
119                // First time running the scheduler: just pick the first thread.
120                if let Some(next_id) = self.ready_queue.pop_front() {
121                    self.current_thread = Some(next_id);
122                    let next_thread = &mut self.threads[next_id];
123                    let stack_ptr = next_thread.stack_ptr;
124
125                    // Since there is no "old" thread yet, we jump straight in.
126                    unsafe {
127                        // Note: We can't use switch_context here because we don't have
128                        // an old_stack to save to. We'll do a manual jump or
129                        // handle this in a bootstrap function.
130                        core::arch::asm!(
131                            "mov rsp, {0}",
132                            "pop r15",
133                            "pop r14",
134                            "pop r13",
135                            "pop r12",
136                            "pop rbx",
137                            "pop rbp",
138                            "ret",
139                            in(reg) stack_ptr,
140                        );
141                    }
142                }
143                return;
144            }
145        };
146
147        // 1. Update current thread state
148        if let Some(thread) = self.threads.get_mut(current_id) {
149            thread.state = ThreadState::Ready;
150            self.ready_queue.push_back(current_id);
151        }
152
153        // 2. Pick next thread
154        if let Some(next_id) = self.ready_queue.pop_front() {
155            self.current_thread = Some(next_id);
156
157            // CR3 Switch: If the next thread is a user process, we must switch address spaces.
158            // (In a full implementation, we would map ThreadId to ProcessId)
159            // For now, we assume kernel threads use the global page table and user processes
160            // have their own. This is a stub for the integration.
161
162            let old_stack_ptr_addr = &mut self.threads[current_id].stack_ptr as *mut usize;
163            let new_stack_ptr = self.threads[next_id].stack_ptr;
164
165            unsafe {
166                switch_context(old_stack_ptr_addr, new_stack_ptr);
167            }
168        }
169    }
