1// VikingOS Kernel — Font Renderer

3// Provides a simple bitmap font for rendering text on the framebuffer.
4
5use crate::drivers::framebuffer::Framebuffer;
6
7/// A simple 8x8 bitmap font.
8/// Each byte in the array represents one row of 8 pixels.
9/// 1 = Foreground, 0 = Background.
10const FONT_8x8: [[u8; 8]; 128] = [
11    [0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00], // 0
12    [0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00], // 1... (placeholder)
13    // We will fill in a few characters for verification.
14    // 'V' (simplified)
15    [
16        0x80, 0x40, 0x20, 0x10, 0x08, 0x04, 0x02, 0x01,
17    ],
18    // 'I'
19    [
20        0xFF, 0x08, 0x08, 0x08, 0x08, 0x08, 0x08, 0xFF,
21    ],
22    // 'K'
23    [
24        0x80, 0x80, 0x80, 0x80, 0x80, 0x80, 0x80, 0x80,
25    ],
26];
27
28/// Render a single character to the framebuffer.
29pub fn draw_char(fb: &mut Framebuffer, x: usize, y: usize, c: char, color: u32) {
30    let glyph_idx = c as usize;
31    if glyph_idx >= FONT_8x8.len() {
32        return;
33    }
34
35    let glyph = &FONT_8x8[glyph_idx];
36
37    for row in 0..8 {
38        let bits = glyph[row];
39        for col in 0..8 {
40            if (bits & (1 << (7 - col))) != 0 {
41                fb.put_pixel(x + col, y + row, color);
42            }
43        }
44    }
45}
46
47/// Render a string of characters to the framebuffer.
48pub fn draw_string(fb: &mut Framebuffer, x: usize, y: usize, s: &str, color: u32) {
49    let mut current_x = x;
50    for c in s.chars() {
51        draw_char(fb, current_x, y, c, color);
52        current_x += 8;
53    }
54}
