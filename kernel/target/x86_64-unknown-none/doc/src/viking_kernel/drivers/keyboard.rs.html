1// VikingOS Kernel — PS/2 Keyboard Driver
2//
3// Handles keyboard input via I/O ports 0x60 and 0x64.
4
5#![allow(dead_code)]
6
7use crate::arch::io::{inb, outb};
8use crate::kprintln;
9
10/// Keyboard controller ports.
11const KBD_DATA_PORT: u16 = 0x60;
12const KBD_STATUS_PORT: u16 = 0x64;
13
14/// A simple mapping from PS/2 scan codes to ASCII.
15/// This is a partial map for demonstration.
16fn scan_code_to_ascii(scancode: u8) -> Option<char> {
17    match scancode {
18        0x10 => Some('q'), 0x11 => Some('w'), 0x12 => Some('e'), 0x13 => Some('r'),
19        0x14 => Some('t'), 0x15 => Some('y'), 0x16 => Some('u'), 0x17 => Some('i'),
20        0x18 => Some('o'), 0x19 => Some('p'), 0x1A => Some('a'), 0x1B => Some('s'),
21        0x1C => Some('d'), 0x1D => Some('f'), 0x1E => Some('g'), 0x1F => Some('h'),
22        0x20 => Some('j'), 0x21 => Some('k'), 0x22 => Some('l'), 0x23 => Some(';'),
23        0x24 => Some('\''), 0x25 => Some('z'), 0x26 => Some('x'), 0x27 => Some('c'),
24        0x28 => Some('v'), 0x29 => Some('b'), 0x2A => Some('n'), 0x2B => Some('m'),
25        0x2C => Some(','), 0x2D => Some('.'), 0x2E => Some('/'), 0x39 => Some(' '),
26        _ => None,
27    }
28}
29
30pub struct Keyboard;
31
32impl Keyboard {
33    /// Initialize the keyboard controller.
34    pub fn init() {
35        // No special initialization needed for basic PS/2 read.
36    }
37
38    /// Read a scancode from the keyboard data port.
39    ///
40    /// # Safety
41    /// Reading from I/O ports can have arbitrary side effects.
42    pub unsafe fn read_scancode() -> Option<u8> {
43        // Check if the output buffer is full (bit 0 of status register)
44        if (inb(KBD_STATUS_PORT) & 0x01) == 0 {
45            return Some(inb(KBD_DATA_PORT));
46        }
47        None
48    }
49
50    /// Handle a keyboard interrupt.
51    pub fn handle_interrupt() {
52        unsafe {
53            if let Some(scancode) = Self::read_scancode() {
54                // Only handle "press" events (scancodes > 0x80 are "release" events)
55                if scancode < 0x80 {
56                    if let Some(c) = scan_code_to_ascii(scancode) {
57                        kprintln!("[Keyboard] Received: {}", c);
58                    } else {
59                        kprintln!("[Keyboard] Received unknown scancode: {:#04x}", scancode);
60                    }
61                }
62            }
63        }
64    }
